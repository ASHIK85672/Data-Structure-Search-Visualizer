Enter values and select Linear Search or Binary Search.

Linear Search

Linear Search checks each element one by one from the beginning until the target is found.

Best: O(1) Worst: O(n)
  1. Start from index 0.
  2. Compare the current value with the target.
  3. If equal, stop and report the position.
  4. Otherwise move to the next value.

Binary Search

Binary Search repeatedly checks the middle value of a sorted array and removes half of the remaining search area.

Best: O(1) Worst: O(log n)
  1. Sort the array.
  2. Find the middle element.
  3. Compare the middle value with the target.
  4. Continue only in the left or right half.

#include <stdio.h>

int main() {
    int n, target, i, found = -1;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    int arr[n];

    printf("Enter %d elements:\n", n);
    for (i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
    }

    printf("Enter element to search: ");
    scanf("%d", &target);

    for (i = 0; i < n; i++) {
        printf("Step %d: Comparing %d with %d\n",
               i + 1, arr[i], target);

        if (arr[i] == target) {
            found = i;
            break;
        }
    }

    if (found != -1)
        printf("Element found at index %d\n", found);
    else
        printf("Element not found\n");

    return 0;
}
#include <stdio.h>

void sortArray(int arr[], int n) {
    int i, j, temp;

    for (i = 0; i < n - 1; i++) {
        for (j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}

int main() {
    int n, target, left, right, mid, i;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    int arr[n];

    printf("Enter %d elements:\n", n);
    for (i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
    }

    sortArray(arr, n);

    printf("Sorted array: ");
    for (i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }

    printf("\nEnter element to search: ");
    scanf("%d", &target);

    left = 0;
    right = n - 1;

    while (left <= right) {
        mid = left + (right - left) / 2;

        printf("\nChecking index %d, value = %d\n",
               mid, arr[mid]);

        if (arr[mid] == target) {
            printf("Element found at index %d\n", mid);
            return 0;
        } else if (arr[mid] < target) {
            printf("Target is greater. Search right half.\n");
            left = mid + 1;
        } else {
            printf("Target is smaller. Search left half.\n");
            right = mid - 1;
        }
    }

    printf("Element not found\n");

    return 0;
}
